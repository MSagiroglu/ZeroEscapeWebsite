tests/e2e/checkout.spec.ts (Page Object Model) 
import { test, expect, type Page } from '@playwright/test';

class CheckoutPage {
  constructor(private page: Page) {}
  open = () => this.page.goto('/upgrade');
  payWithTestCard = () => this.page.getByRole('button', { name: 'Pay with test card' }).click();
}

test('plan becomes Pro even when webhook is delayed', async ({ page }) => {
  await page.route('**/api/billing/status', async (route) => {
    await new Promise((r) => setTimeout(r, 4000)); // simulate slow webhook
    await route.continue();
  });
  const checkout = new CheckoutPage(page);
  await checkout.open();
  await checkout.payWithTestCard();
  await expect(page.getByTestId('plan-badge')).toHaveText('Pro', { timeout: 15_000 });
});
